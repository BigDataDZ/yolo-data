10: function formatTemporalDisplay(value, timeGrain) {
11:   const text = String(value ?? '');
12:   const match = text.match(/^(\d{4})-(\d{2})-(\d{2})(?:[T\s].*)?$/);
13:   if (!match) {
14:     return text;
15:   }
16:   const [, year, month] = match;
17:   switch (String(timeGrain ?? '').toUpperCase()) {

417: function periodComparisonFindings(
418:   columns,
419:   rows,
420:   contract,
421:   metricDefinitions = [],
422:   lexicon = null,
423: ) {
424:   const periodKey = '__period';
425:   if (!columns.some((column) => columnKey(column) === periodKey)) {
426:     return [];
427:   }
428:   const availablePeriods = [...new Set(

434:       window.label || window.expression || `期间${index + 1}`

437:     ? contract.calculation.baseWindowIndex

439:   const compareIndex = Number.isInteger(contract?.calculation?.compareWindowIndex)
440:     ? contract.calculation.compareWindowIndex
441:     : 1;
442:   const currentPeriod = periodOrder[baseIndex] ?? availablePeriods[0];
443:   const previousPeriod = periodOrder[compareIndex] ?? availablePeriods[1];
444:   if (!currentPeriod || !previousPeriod) {
445:     return [];
446:   }
447: 
448:   const ratePattern = lexiconPattern((lexicon ?? getPlatformLexicon()).rateTerms);
449:   const definitions = new Map(
450:     (metricDefinitions ?? []).map((definition) => [
451:       String(definition.key ?? definition.field ?? ''),
452:       definition,
453:     ]),
454:   );
455:   const metricColumns = (contract?.metrics ?? [])
456:     .map((key) => findColumn(columns, key))
457:     .filter((column) => column && isNumericColumn(column));
458:   const dimensionColumn = (contract?.dimensions ?? [])
459:     .map((key) => findColumn(columns, key))

469:     const definition = definitions.get(metricKey) ?? {};
470:     const metricLabel = columnLabel(metricColumn)
471:       || definition.label
472:       || metricKey;
473:     const currentTotal = numericTotal(rows, metricKey, currentPeriod);
474:     const previousTotal = numericTotal(rows, metricKey, previousPeriod);
475:     const delta = currentTotal - previousTotal;
476:     const rate = previousTotal !== 0
477:       ? delta / Math.abs(previousTotal)
478:       : null;
479:     const direction = delta > 0
480:       ? '增长'
481:       : delta < 0
482:         ? '下降'
483:         : '持平';
484:     const directionPrefix = delta > 0 ? '+' : '';
485:     const rateText = rate === null
486:       ? ''
487:       : `，同比${direction} ${directionPrefix}${(Math.abs(rate) * 100).toFixed(1)}%`;
488:     const amountText = Number.isFinite(delta)
489:       ? `，${delta > 0 ? '增加' : delta < 0 ? '减少' : '变动'} ${
490:         formatColumnValue(Math.abs(delta), metricColumn)
491:       }`
492:       : '';
493:     const statement = `${metricLabel}${rateText}${amountText}（${
494:       currentPeriod
495:     } ${formatColumnValue(currentTotal, metricColumn)}，${
496:       previousPeriod
497:     } ${formatColumnValue(previousTotal, metricColumn)}）`;
498: 
499:     const detail = [];

501:       const dimensionKey = columnKey(dimensionColumn);
502:       const dimensionLabel = columnLabel(dimensionColumn) || dimensionKey;
503:       const grouped = new Map();
504:       for (const row of rows ?? []) {
505:         const period = String(row?.[periodKey] ?? '');
506:         if (![currentPeriod, previousPeriod].includes(period)) {
507:           continue;
508:         }
509:         const value = Number(row?.[metricKey]);
510:         if (!Number.isFinite(value)) {
511:           continue;
512:         }
513:         const name = String(row?.[dimensionKey] ?? '').trim();
514:         if (!name) {
515:           continue;
516:         }
517:         const item = grouped.get(name) ?? { current: 0, previous: 0 };
518:         if (period === currentPeriod) {
519:           item.current += value;
520:         } else {
521:           item.previous += value;
522:         }
523:         grouped.set(name, item);
524:       }
525:       const changes = [...grouped.entries()]
526:         .map(([name, item]) => ({
527:           name,
528:           delta: item.current - item.previous,
529:         }))
530:         .filter((item) => Number.isFinite(item.delta))
531:         .sort((left, right) => right.delta - left.delta);
532:       const topGrowth = changes
533:         .filter((item) => item.delta > 0)
534:         .slice(0, 3);

537:           topGrowth.map((item) => (
538:             `${item.name}（+${formatColumnValue(item.delta, metricColumn)}）`
539:           )).join('、')
540:         }`);
541:       } else {
542:         const largest = changes.slice(0, 3);
543:         if (largest.length > 0) {
544:           detail.push(`当前没有${dimensionLabel}实现增长，变化靠前的为：${
545:             largest.map((item) => (
546:               `${item.name}（${formatColumnValue(item.delta, metricColumn)}）`
547:             )).join('、')
548:           }`);
549:         }
550:       }
551:     }
552:     return {
553:       statement,
554:       details: detail,
555:       metricKey,
556:       metricLabel,
557:       currentPeriod,
558:       previousPeriod,
559:       currentTotal,
560:       previousTotal,
561:       delta,
562:       rate,
563:     };
564:   });

616:     .find((column) => (

618:       && columnKey(column) !== periodKey
619:       && !isDateColumn(column)
620:       && !isNumericColumn(column)
621:     ));
622:   const metricKey = selectSortMetric(columns, contract);
623:   if (!dimensionColumn || !metricKey) {
624:     return [];
625:   }
626:   const dimensionKey = columnKey(dimensionColumn);
627:   const periods = [...new Set(rows.map((row) => String(row?.[periodKey] ?? '')).filter(Boolean))];
628:   const periodOrder = contract?.periodOrder ?? [];
629:   const baseIndex = Number.isInteger(contract?.calculation?.baseWindowIndex)
630:     ? contract.calculation.baseWindowIndex
631:     : 0;
632:   const compareIndex = Number.isInteger(contract?.calculation?.compareWindowIndex)
633:     ? contract.calculation.compareWindowIndex
634:     : 1;
635:   const currentPeriod = periodOrder[baseIndex] ?? periods[0];
636:   const previousPeriod = periodOrder[compareIndex] ?? periods[1];
637:   if (!currentPeriod || !previousPeriod) {
638:     return [];
639:   }
640: 
641:   const grouped = new Map();
642:   for (const row of rows) {
643:     const dimensionValue = String(row?.[dimensionKey] ?? '');
644:     if (!dimensionValue) {
645:       continue;
646:     }
647:     const current = grouped.get(dimensionValue) ?? {
648:       current: 0,
649:       previous: 0,
650:     };
651:     const value = Number(row?.[metricKey]);
652:     if (!Number.isFinite(value)) {
653:       continue;
654:     }
655:     if (String(row?.[periodKey]) === String(currentPeriod)) {
656:       current.current += value;
657:     } else if (String(row?.[periodKey]) === String(previousPeriod)) {
658:       current.previous += value;
659:     }
660:     grouped.set(dimensionValue, current);
661:   }
662: 
663:   const changes = [...grouped.entries()].map(([dimensionValue, values]) => ({
664:     dimensionValue,
665:     ...values,
666:     delta: values.current - values.previous,
667:   })).filter((item) => item.current !== 0 || item.previous !== 0);
668:   if (changes.length === 0) {
669:     return [];
670:   }
671:   changes.sort((left, right) => Math.abs(right.delta) - Math.abs(left.delta));
672:   const totalDelta = changes.reduce((sum, item) => sum + item.delta, 0);
673:   const metricColumn = findColumn(columns, metricKey);
674:   const metricLabel = columnLabel(metricColumn) || metricKey;
675:   const dimensionLabel = columnLabel(dimensionColumn) || dimensionKey;
676:   const rateMetric = ratePattern?.test(metricLabel) ?? false;
677:   return changes.slice(0, 5).map((item) => {
678:     const contribution = !rateMetric && Math.abs(totalDelta) > Number.EPSILON
679:       ? `，对整体变化贡献 ${(item.delta / totalDelta * 100).toFixed(1)}%`
680:       : '';
681:     return `${dimensionLabel}「${item.dimensionValue}」${metricLabel}变化 ${
682:       item.delta >= 0 ? '+' : ''
683:     }${formatColumnValue(item.delta, metricColumn)}${contribution}`;
684:   });
685: }

687: function rankingFinding(question, columns, rows, contract) {
688:   if (!RANKING_PATTERN.test(String(question ?? '')) || rows.length === 0) {
689:     return '';
690:   }
691:   const metricKey = selectSortMetric(columns, contract);
692:   if (!metricKey) {
693:     return '';
694:   }
695:   const dimensionKey = (contract?.dimensions ?? [])
696:     .map((key) => findColumn(columns, key))
697:     .find((column) => column && !isNumericColumn(column) && !isDateColumn(column));
698:   const metricColumn = findColumn(columns, metricKey);
699:   const metricLabel = columnLabel(metricColumn) || metricKey;
700:   if (!dimensionKey) {
701:     return `最高/最低值为 ${formatColumnValue(rows[0]?.[metricKey], metricColumn)}`;
702:   }
703:   const dimensionLabel = columnLabel(dimensionKey);
704:   return `${dimensionLabel}首位为 ${formatNumber(rows[0]?.[columnKey(dimensionKey)])}，${
705:     metricLabel} ${formatColumnValue(rows[0]?.[metricKey], metricColumn)}`;
706: }
